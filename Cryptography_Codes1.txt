#include <iostream>
using namespace std;
int gcd (int r1, int r2){
    while( r2 != 0){
        int r = r1 % r2;
        r1 = r2;
        r2 = r;
    }
    return r1;
}

int main(){
    int p1 = 17, p2 = 11, e = 4, d = 1, m = 88;
    int n = p1*p2;
    int phi = (p1 - 1)*(p2 - 1);
    
    while(e < phi){
        if(gcd(e,phi) == 1){
            break;
        }
        else
            e++;
    }
    
    while(true){
        if((d*e) % phi == 1 ){
            break;
        }
        else
            d++;
    }
    long long c = 1;
    for(int i=0; i<e; i++){
        c = (c * m) % n;
    }
    
    cout<<"The First Prime number is "<<p1<<endl;
    cout<<"The Second Prime number is "<<p2<<endl;
    cout<<"The Modular is "<<n<<endl;
    cout<<"The Euler Toitent is "<<phi<<endl;
    cout<<"The Public key is "<<e<<endl;
    cout<<"The Private key is  "<<d<<endl;
    cout<<"The Encrypted Text is "<<c<<endl;
    return 0;
}

long long power(long long base, long long exp, long long n){
    int result = 1;
    while (exp > 0){
        if(exp % 2 == 1){
            result = (result * base) % n;
        }
        exp = exp/2;
        base = (base * base) % n;
    }
    return result;
}

#include <iostream>
using namespace std;

int main() {
    int message = 88;
    int privateKey = 23;
    int publicKey = 23;
    int hash = message % 97;
    int signature = hash * privateKey;
    int receivedHash = signature / publicKey;
    cout << "Original Message: " << message << endl;
    cout << "Message Hash: " << hash << endl;
    cout << "Digital Signature: " << signature << endl;
    cout << "Verified Hash: " << receivedHash << endl;
    if (hash == receivedHash) {
        cout << "Digital Signature is VALID";
    }
    else {
        cout << "Digital Signature is INVALID";
    }
    return 0;
}

int main(){
    string text = "ABCDEFGHIJK";
    string key = "231";
    int column = key.length();
    while(text.length() % column != 0){
        text += 'X';
    }
    int row = text.length()/column;
    char grid[row][column];
    
    int i=0;
    for(int r=0; r<row; r++){
        for(int c=0; c<column; c++){
            grid[r][c] = text[i];
            i++;
        }
    }
    
    string enc = "";
     for (int k = 1; k <= column; k++) {
        for (int c = 0; c < column; c++) {
            if (key[c] - '0' == k) {
                for (int r = 0; r < row; r++) {
                    enc += grid[r][c];
                }
            }
        }
    }
    cout<<"CipherText is "<<enc<<endl;
    return 0;
}
